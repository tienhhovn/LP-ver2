            const close = seedPrice + trend + randomVolatility;

            const open = i === 0 ? seedPrice - 100 : candles[i - 1].close;

            const high = Math.max(open, close) + (Math.abs(Math.sin(i)) * 180 + 50);

        function resizeCanvas() {

            const dpr = window.devicePixelRatio || 1;

            const rect = canvas.getBoundingClientRect();

            canvas.width = rect.width * dpr;

            canvas.height = rect.height * dpr;

            ctx.scale(dpr, dpr);

            drawChart();

        }


        function drawChart() {

            const width = canvas.clientWidth;

            const height = canvas.clientHeight;

            ctx.clearRect(0, 0, width, height);


            const paddingLeft = 40;

            const paddingRight = 60;

            const paddingTop = 40;

            const paddingBottom = 40;


            const chartWidth = width - paddingLeft - paddingRight;

            const chartHeight = height - paddingTop - paddingBottom;


            // Tìm min max của nến để scale đồ thị chính xác

            let minVal = Infinity;

            let maxVal = -Infinity;

            candles.forEach(c => {

                if (c.low < minVal) minVal = c.low;

                if (c.high > maxVal) maxVal = c.high;

            });


            // Cộng thêm khoảng thở

            minVal -= 200;

            maxVal += 200;


            const getX = (index) => paddingLeft + (index / (totalCandles - 1)) * chartWidth;

            const getY = (val) => paddingTop + chartHeight - ((val - minVal) / (maxVal - minVal)) * chartHeight;


            // 1. Vẽ các đường kẻ lưới phụ (Grid lines)

            ctx.strokeStyle = 'rgba(255, 255, 255, 0.03)';

            ctx.lineWidth = 1;

            for (let i = 1; i < 5; i++) {

                const y = paddingTop + (chartHeight / 5) * i;

                ctx.beginPath();

                ctx.moveTo(paddingLeft, y);

                ctx.lineTo(width - paddingRight, y);

                ctx.stroke();

                ctx.fillStyle = 'rgba(34, 197, 94, 0.06)';

                ctx.fillRect(paddingLeft, paddingTop + chartHeight * 0.78, chartWidth, chartHeight * 0.22);

                ctx.moveTo(paddingLeft, paddingTop + chartHeight * 0.78);

            // 3. Vẽ nến Nhật (Candlesticks)

                ctx.beginPath();

                ctx.moveTo(x, yHigh);

                ctx.lineTo(x, yLow);

                ctx.stroke();


                // Vẽ thân nến

                const rectX = x - candleWidth / 2;

                const rectY = Math.min(yOpen, yClose);

        // Lightweight Markdown parser to format LLM bold texts, lists, headers and codes

        function parseMarkdown(text) {

            let html = text;

            

            // Code blocks

            html = html.replace(/```([\s\S]*?)```/g, '<pre class="bg-slate-950/80 text-emerald-400 p-3 rounded-lg text-xs font-mono my-2 overflow-x-auto border border-slate-800/80">$1</pre>');

            

            // Inline code

            html = html.replace(/`([^`]+)`/g, '<code class="bg-slate-950 px-1.5 py-0.5 rounded text-sky-400 font-mono text-xs border border-white/5">$1</code>');

            

            // Bold

            html = html.replace(/\*\*([^*]+)\*\*/g, '<strong class="font-bold text-white">$1</strong>');

            

            // Headers

            html = html.replace(/^### (.*$)/gim, '<h5 class="text-sm font-bold text-sky-400 mt-3 mb-1">$1</h5>');

            html = html.replace(/^## (.*$)/gim, '<h4 class="text-base font-bold text-sky-300 mt-4 mb-2">$1</h4>');

            html = html.replace(/^# (.*$)/gim, '<h3 class="text-lg font-bold text-white mt-5 mb-3">$1</h3>');

            

            // Bullet lists

            html = html.replace(/^\s*-\s+(.*$)/gim, '<li class="list-disc list-inside text-slate-300 ml-2 mt-1">$1</li>');

            html = html.replace(/^\s*\*\s+(.*$)/gim, '<li class="list-disc list-inside text-slate-300 ml-2 mt-1">$1</li>');

            

            // Newlines

            html = html.replace(/\n/g, '<br>');

            

            return html;

        }


        // Core fetch operation with exponential backoff handler

        async function fetchWithBackoff(url, options, retries = 3, delay = 1000) {

            try {

                const response = await fetch(url, options);

                if (response.status === 429 && retries > 0) {

                    // Throttled by Google API -> trigger exponential backoff

                    await new Promise(resolve => setTimeout(resolve, delay));

                    return await fetchWithBackoff(url, options, retries - 1, delay * 2);

                }

                return response;

            } catch (error) {

                if (retries > 0) {

                    await new Promise(resolve => setTimeout(resolve, delay));

                    return await fetchWithBackoff(url, options, retries - 1, delay * 2);

                }

                throw error;

            }

        }

        function removeLoadingIndicator() {

        // Handle chat submit

        aiChatForm.addEventListener('submit', async (e) => {

            e.preventDefault();

            const messageText = aiChatInput.value.trim();

            if (!messageText) return;


            // Clear input field

            aiChatInput.value = '';


            // Render User's message

            appendMessage('user', messageText);


            // Display loading bubble

            appendLoadingIndicator();

            sourcesContainer.classList.add('hidden');

            sourcesList.innerHTML = '';


            try {

                const responseData = await askGemini(messageText, searchGroundingEnabled);

                removeLoadingIndicator();


                const candidate = responseData?.candidates?.[0];

                if (candidate && candidate.content?.parts?.[0]?.text) {

                    const generatedText = candidate.content.parts[0].text;

                    

                    // Display AI message

                    appendMessage('ai', generatedText, true);


                    // Extract and display search grounding citations if present

                    const groundingMetadata = candidate.groundingMetadata;

                    if (groundingMetadata && groundingMetadata.groundingAttributions) {

                        const attributions = groundingMetadata.groundingAttributions

                            .map(attr => ({

                                uri: attr.web?.uri,

                                title: attr.web?.title || attr.web?.uri,

                            }))

                            .filter(src => src.uri && src.title);


                        if (attributions.length > 0) {

                            sourcesList.innerHTML = '';

                            attributions.forEach(source => {

                                const chip = document.createElement('a');

                                chip.href = source.uri;

                                chip.target = '_blank';

                                chip.className = 'text-[10px] bg-sky-500/10 hover:bg-sky-500/20 border border-sky-500/20 text-sky-400 px-2 py-0.5 rounded-full flex items-center gap-1 max-w-[150px] truncate transition';

                                chip.innerHTML = `<i class="fa-solid fa-arrow-up-right-from-square text-[8px]"></i> ${source.title}`;

                                sourcesList.appendChild(chip);

                            });

                            sourcesContainer.classList.remove('hidden');

                        }

                    }

                } else {

                    appendMessage('ai', '⚠️ Xin lỗi, mô hình đã trả về cấu trúc dữ liệu không hợp lệ. Vui lòng thử lại.');

                }

            } catch (err) {

                removeLoadingIndicator();

                console.error("Gemini API Error:", err);

                appendMessage('ai', `❌ **Đã xảy ra lỗi:** \`${err.message}\`. Vui lòng xác minh kết nối internet hoặc kiểm tra lại API key nếu bạn đã thiết lập thủ công.`);

            }

        });
